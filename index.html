mask.ts 
// Recall modes: first letters → every 2nd word → fully hidden
export function maskWords(content: string, level: MemoryLevel,
    revealed: readonly boolean[] = []): MaskedWord[] {
  return content
    .trim()
    .split(/\s+/u)
    .filter(Boolean)
    .map((word, index) => {
      const hide = level === 'initials' ? true
        : level === 'alternate' ? index % 2 === 1 : true
      if (!hide || revealed[index]) return { text: word, masked: false }
      return { text: hideLetters(word, level === 'initials' ? 1 : 0),
               masked: true }
    })
}

progression.ts 
// Double progression: level up only when target met with reps in reserve
export function judge(state: AppState, id: ExerciseId,
    rows: SetResult[], effort: EffortKey): Verdict {
  const prescribed = plan(state, id);
  const done = rows.filter((r) => r.reps > 0);
  if (!done.length) return 'none';

  const met =
    done.length >= prescribed.length &&
    done.every((r, i) => r.reps >= (prescribed[i]?.reps ?? 0));
  const short = done.some((r, i) => {
    const target = prescribed[i]?.reps;
    return target !== undefined && r.reps <= target - 2 * step(id);
  });
  const reserve = effort !== 'max';

  if (met && reserve) return 'up';
  if (met && !reserve) return 'hold-max';
  if (short) return 'down';
  return 'hold';
}

scoring.ts 
// Reverse-keyed Likert sums, rescaled per scale onto a normative 1–100
export function scoreLikert(items: ScoredItem[], answers: Answers,
    scaleDef: Scale): LikertScores {
  const raw: Record<string, number> = {};
  const counts: Record<string, number> = {};
  for (const item of items) {
    counts[item.scale] = (counts[item.scale] ?? 0) + 1;
    raw[item.scale] ??= 0;
    const given = answers[item.id];
    const isNumber = typeof given === "number" && Number.isFinite(given);
    const value = isNumber ? clamp(given, scaleDef.min, scaleDef.max)
      : (scaleDef.min + scaleDef.max) / 2;
    raw[item.scale] += item.reverse ? flip(value, scaleDef) : value;
  }

search.sql 
-- Hybrid RRF: semantic + full-text fused by rank
WITH semantic AS (
  SELECT id, ROW_NUMBER() OVER (ORDER BY embedding <=> $1) rk
  FROM verses ORDER BY embedding <=> $1 LIMIT 60
), lexical AS (
  SELECT id, ROW_NUMBER() OVER (ORDER BY ts_rank(fts, q) DESC) rk
  FROM verses, websearch_to_tsquery($2) q WHERE fts @@ q LIMIT 60
)
SELECT id, SUM(1.0/(60+rk)) score
FROM (SELECT * FROM semantic UNION ALL SELECT * FROM lexical) t
GROUP BY id ORDER BY score DESC LIMIT $3
